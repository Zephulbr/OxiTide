# Oneko Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an opt-in oneko cat that chases the cursor through the OxiTide window, can be dragged, double-clicked to sleep on the now-playing seek bar, and toggled/persisted from Settings.

**Architecture:** A DOM-free engine (`ui/src/lib/oneko.ts`) ports oneko.js + spicetify-oneko behavior with an injectable RNG; a thin Svelte shell (`Neko.svelte`) owns RAF/listeners/DOM writes; the enable flag travels the existing `start_in_mini` config path (Rust config → command → handler → `api.ts` → appearance singleton → Settings checkbox).

**Tech Stack:** Rust (serde TOML config, Tauri 2 commands), TypeScript, Svelte 5 runes, vitest, Vite asset imports, PowerShell 5.1 for the asset download.

**Spec:** `docs/superpowers/specs/2026-09-29-oxitide-oneko-design.md`

## Global Constraints

- **Do not commit.** The owner commits manually (session agreement); each task ends with its verification commands instead of a commit step.
- No new dependencies in `ui/package.json` or `Cargo.toml`.
- No comments in new source code.
- Styling consumes semantic tokens only; no `data-mode` branching in components (AGENTS.md).
- IPC args are camelCase from the UI; Rust command params stay snake_case; `ui/src/lib/api.ts` interfaces are kept in sync by hand.
- Windows PowerShell 5.1; run all commands from the repo root `C:\Users\V\Documents\Opencode\OxiTide`.
- Sprite sheet geometry is fixed: 256×128 gif, 8×4 cells of 32px, CSS `background-position` uses negative offsets (`col * -32px`, `row * -32px`).
- Reference constants are exact: speed 10px/tick, tick 100ms, idle radius 48px, sleep offsets `x = perch.right - 16`, `y = perch.top - 8`.

## Review Focus

The spec implies these inputs/failure modes; each is pinned to a test in the task that owns it.

1. A position saved in a larger window must come back on-screen (Task 3: `clampPosition` / `setPosition` bounds tests).
2. The seek bar can vanish while she is asleep (track ends) — she must settle in place, not freeze or drift (Task 5: "sleeps in place when the perch disappears").
3. Mouse moves while she is commanded to sleep must not retarget her, or she will never settle on the perch (Task 5: "walks to the perch and settles" with a `setMouse(0, 0)` call).
4. Dragging her while asleep must end with her back on the perch, asleep (Task 5: "returns to the perch after being dragged while asleep").
5. Dragging must clamp at the window edges so she cannot be lost off-screen (Task 5: "clamps drag inside the viewport").

```ts
export const NEKO_SIZE = 32;
export const NEKO_SPEED = 10;
export const TICK_MS = 100;
export const IDLE_RADIUS = 48;

export interface Sprite {
  col: number;
  row: number;
}

export interface NekoFrame {
  x: number;
  y: number;
  sprite: Sprite;
}

export interface PerchRect {
  right: number;
  top: number;
}

export interface NekoEngine {
  setMouse(x: number, y: number): void;
  setViewport(width: number, height: number): void;
  setPosition(x: number, y: number): NekoFrame;
  tick(): NekoFrame;
}

type SpriteName =
  | "idle"
  | "alert"
  | "tired"
  | "sleeping"
  | "scratchSelf"
  | "scratchWallN"
  | "scratchWallS"
  | "scratchWallE"
  | "scratchWallW"
  | "N"
  | "NE"
  | "E"
  | "SE"
  | "S"
  | "SW"
  | "W"
  | "NW";

const SPRITES: Record<SpriteName, Sprite[]> = {
  idle: [{ col: 3, row: 3 }],
  alert: [{ col: 7, row: 3 }],
  scratchSelf: [{ col: 5, row: 0 }, { col: 6, row: 0 }, { col: 7, row: 0 }],
  scratchWallN: [{ col: 0, row: 0 }, { col: 0, row: 1 }],
  scratchWallS: [{ col: 7, row: 1 }, { col: 6, row: 2 }],
  scratchWallE: [{ col: 2, row: 2 }, { col: 2, row: 3 }],
  scratchWallW: [{ col: 4, row: 0 }, { col: 4, row: 1 }],
  tired: [{ col: 3, row: 2 }],
  sleeping: [{ col: 2, row: 0 }, { col: 2, row: 1 }],
  N: [{ col: 1, row: 2 }, { col: 1, row: 3 }],
  NE: [{ col: 0, row: 2 }, { col: 0, row: 3 }],
  E: [{ col: 3, row: 0 }, { col: 3, row: 1 }],
  SE: [{ col: 5, row: 1 }, { col: 5, row: 2 }],
  S: [{ col: 6, row: 3 }, { col: 7, row: 2 }],
  SW: [{ col: 5, row: 3 }, { col: 6, row: 1 }],
  W: [{ col: 4, row: 2 }, { col: 4, row: 3 }],
  NW: [{ col: 1, row: 0 }, { col: 1, row: 1 }],
};

export function clampPosition(
  x: number,
  y: number,
  width: number,
  height: number,
): { x: number; y: number } {
  const half = NEKO_SIZE / 2;
  return {
    x: Math.min(Math.max(x, half), Math.max(half, width - half)),
    y: Math.min(Math.max(y, half), Math.max(half, height - half)),
  };
}

export function createNekoEngine(opts: { rng?: () => number } = {}): NekoEngine {
  const rng = opts.rng ?? Math.random;
  let x = 32;
  let y = 32;
  let mouseX = 0;
  let mouseY = 0;
  let width = 0;
  let height = 0;
  let frameCount = 0;
  let idleTime = 0;

  function spriteFor(name: SpriteName, frame: number): Sprite {
    const set = SPRITES[name];
    return set[((frame % set.length) + set.length) % set.length];
  }

  function idleFrame(): NekoFrame {
    idleTime += 1;
    return { x, y, sprite: spriteFor("idle", 0) };
  }

  function directionFor(diffX: number, diffY: number, distance: number): SpriteName {
    let direction = diffY / distance > 0.5 ? "N" : "";
    direction += diffY / distance < -0.5 ? "S" : "";
    direction += diffX / distance > 0.5 ? "W" : "";
    direction += diffX / distance < -0.5 ? "E" : "";
    return direction as SpriteName;
  }

  function setMouse(mx: number, my: number) {
    mouseX = mx;
    mouseY = my;
  }

  function setViewport(w: number, h: number) {
    width = w;
    height = h;
    const clamped = clampPosition(x, y, width, height);
    x = clamped.x;
    y = clamped.y;
  }

  function setPosition(px: number, py: number): NekoFrame {
    const clamped = clampPosition(px, py, width, height);
    x = clamped.x;
    y = clamped.y;
    return { x, y, sprite: spriteFor("idle", 0) };
  }

  function tick(): NekoFrame {
    frameCount += 1;
    const diffX = x - mouseX;
    const diffY = y - mouseY;
    const distance = Math.hypot(diffX, diffY);
    if (distance < NEKO_SPEED || distance < IDLE_RADIUS) return idleFrame();
    const sprite = spriteFor(directionFor(diffX, diffY, distance), frameCount);
    x -= (diffX / distance) * NEKO_SPEED;
    y -= (diffY / distance) * NEKO_SPEED;
    const clamped = clampPosition(x, y, width, height);
    x = clamped.x;
    y = clamped.y;
    return { x, y, sprite };
  }

  return { setMouse, setViewport, setPosition, tick };
}
```

```ts
  function resetIdleAnimation() {
    idleAnimation = null;
    idleAnimationFrame = 0;
  }

  function idleFrame(): NekoFrame {
    idleTime += 1;
    if (idleTime > 10 && Math.floor(rng() * 200) === 0 && idleAnimation === null) {
      const options: SpriteName[] = ["sleeping", "scratchSelf"];
      if (x < NEKO_SIZE) options.push("scratchWallW");
      if (y < NEKO_SIZE) options.push("scratchWallN");
      if (x > width - NEKO_SIZE) options.push("scratchWallE");
      if (y > height - NEKO_SIZE) options.push("scratchWallS");
      idleAnimation = options[Math.floor(rng() * options.length)];
    }
    let sprite = spriteFor("idle", 0);
    switch (idleAnimation) {
      case "sleeping":
        if (idleAnimationFrame < 8) {
          sprite = spriteFor("tired", 0);
          break;
        }
        sprite = spriteFor("sleeping", Math.floor(idleAnimationFrame / 4));
        if (idleAnimationFrame > 192) resetIdleAnimation();
        break;
      case "scratchWallN":
      case "scratchWallS":
      case "scratchWallE":
      case "scratchWallW":
      case "scratchSelf":
        sprite = spriteFor(idleAnimation, idleAnimationFrame);
        if (idleAnimationFrame > 9) resetIdleAnimation();
        break;
      default:
        return { x, y, sprite };
    }
    idleAnimationFrame += 1;
    return { x, y, sprite };
  }
```

```svelte
<script lang="ts">
  import onekoUrl from "../../assets/oneko/oneko.gif";
  import {
    clampPosition,
    createNekoEngine,
    NEKO_SIZE,
    TICK_MS,
    type NekoFrame,
    type PerchRect,
  } from "../oneko";

  const STORAGE_KEY = "oxitide:oneko";
  const SAVE_INTERVAL_MS = 1000;

  let el: HTMLDivElement | undefined;
  const engine = createNekoEngine();

  function loadPosition(): { x: number; y: number } | null {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (!raw) return null;
      const parsed = JSON.parse(raw) as { x?: unknown; y?: unknown };
      if (typeof parsed.x !== "number" || typeof parsed.y !== "number") return null;
      return { x: parsed.x, y: parsed.y };
    } catch {
      return null;
    }
  }

  function perchRect(): PerchRect | null {
    const perch = document.querySelector<HTMLElement>("[data-oneko-perch]");
    if (!perch) return null;
    const rect = perch.getBoundingClientRect();
    if (rect.width === 0 || rect.height === 0) return null;
    return { right: rect.right, top: rect.top };
  }

  $effect(() => {
    const node = el;
    if (!node) return;
    let lastFrame: NekoFrame = { x: NEKO_SIZE, y: NEKO_SIZE, sprite: { col: 3, row: 3 } };
    let lastSave = performance.now();

    function save() {
      try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify({ x: lastFrame.x, y: lastFrame.y }));
      } catch {
        return;
      }
    }

    function apply(frame: NekoFrame) {
      lastFrame = frame;
      node.style.left = `${frame.x - NEKO_SIZE / 2}px`;
      node.style.top = `${frame.y - NEKO_SIZE / 2}px`;
      node.style.backgroundPosition = `${-frame.sprite.col * NEKO_SIZE}px ${-frame.sprite.row * NEKO_SIZE}px`;
      const now = performance.now();
      if (now - lastSave >= SAVE_INTERVAL_MS) {
        lastSave = now;
        save();
      }
    }

    engine.setViewport(window.innerWidth, window.innerHeight);
    const saved = loadPosition();
    if (saved) {
      const pos = clampPosition(saved.x, saved.y, window.innerWidth, window.innerHeight);
      apply(engine.setPosition(pos.x, pos.y));
    } else {
      apply(engine.tick());
    }

    let raf = 0;
    let lastTick = 0;
    const loop = (now: number) => {
      if (now - lastTick >= TICK_MS) {
        lastTick = now;
        apply(engine.tick());
      }
      raf = requestAnimationFrame(loop);
    };
    raf = requestAnimationFrame(loop);

    const onMouseMove = (event: MouseEvent) => engine.setMouse(event.clientX, event.clientY);
    const onResize = () => {
      engine.setViewport(window.innerWidth, window.innerHeight);
      engine.setPerch(perchRect());
    };

    let dragCleanup: (() => void) | null = null;
    const onMouseDown = (event: MouseEvent) => {
      if (event.button !== 0) return;
      event.preventDefault();
      engine.startDrag(event.clientX, event.clientY);
      node.style.cursor = "grabbing";
      const onDragMove = (moveEvent: MouseEvent) => {
        apply(engine.dragTo(moveEvent.clientX, moveEvent.clientY));
      };
      const onDragUp = () => {
        engine.endDrag();
        node.style.cursor = "";
        window.removeEventListener("mousemove", onDragMove);
        window.removeEventListener("mouseup", onDragUp);
        dragCleanup = null;
        save();
      };
      window.addEventListener("mousemove", onDragMove);
      window.addEventListener("mouseup", onDragUp);
      dragCleanup = () => {
        window.removeEventListener("mousemove", onDragMove);
        window.removeEventListener("mouseup", onDragUp);
      };
    };
    const onDblClick = () => engine.toggleSleep(perchRect());

    window.addEventListener("mousemove", onMouseMove);
    window.addEventListener("resize", onResize);
    node.addEventListener("mousedown", onMouseDown);
    node.addEventListener("dblclick", onDblClick);

    return () => {
      cancelAnimationFrame(raf);
      window.removeEventListener("mousemove", onMouseMove);
      window.removeEventListener("resize", onResize);
      node.removeEventListener("mousedown", onMouseDown);
      node.removeEventListener("dblclick", onDblClick);
      dragCleanup?.();
      save();
    };
  });
</script>
```
